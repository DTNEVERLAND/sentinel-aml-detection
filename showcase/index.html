-- ─────────────────────────────────────────────────────────────────────────────
-- Pass-Through Layering Detection
-- Pairs ADA / XRP deposits with fiat withdrawals from the SAME account.
-- Flag fires when funds hit fiat < 2h after deposit AND > 95% of deposit value.
-- Owner: Risk & Compliance Engineering
-- Cadence: every 5 minutes (Airflow DAG: aml_tripwires)
-- ─────────────────────────────────────────────────────────────────────────────

WITH alt_deposits AS (
  -- Step 1. Confirmed inbound transfers of volatile alts in the last 24h.
  SELECT
    user_id,
    txn_id      AS deposit_txn,
    asset       AS deposit_asset,
    amount_usd  AS deposit_value,
    created_at  AS deposit_ts
  FROM ledger.deposits
  WHERE asset IN ('ADA', 'XRP')
    AND status = 'confirmed'
    AND created_at >= NOW() - INTERVAL '24 hours'
),

fiat_withdrawals AS (
  -- Step 2. Pending or processing fiat off-ramps in the same window.
  SELECT
    user_id,
    txn_id      AS withdrawal_txn,
    asset       AS fiat_asset,
    amount_usd  AS withdrawal_value,
    created_at  AS withdrawal_ts
  FROM ledger.withdrawals
  WHERE asset IN ('MYR', 'IDR', 'ZAR', 'EUR', 'GBP')
    AND status IN ('pending', 'processing')
    AND created_at >= NOW() - INTERVAL '24 hours'
),

paired_flows AS (
  -- Step 3. Self-join: pair each deposit with any later fiat exit by the same user.
  SELECT
    d.user_id,
    d.deposit_asset,
    f.fiat_asset,
    d.deposit_value,
    f.withdrawal_value,
    EXTRACT(EPOCH FROM (f.withdrawal_ts - d.deposit_ts)) / 60.0
                                                            AS minutes_elapsed,
    ROUND(f.withdrawal_value / NULLIF(d.deposit_value, 0), 4)
                                                            AS withdrawal_ratio
  FROM alt_deposits d
  JOIN fiat_withdrawals f
    ON  d.user_id = f.user_id
    AND f.withdrawal_ts > d.deposit_ts
)

-- Step 4. Tripwire: instant conversion AND near-total drain → flag.
SELECT
  user_id,
  deposit_asset,
  fiat_asset,
  deposit_value,
  withdrawal_value,
  minutes_elapsed,
  withdrawal_ratio,
  'PASS_THROUGH_LAYERING' AS risk_flag
FROM paired_flows
WHERE minutes_elapsed < 120      -- under 2h
  AND withdrawal_ratio > 0.95   -- >95% of deposit value cashed out
ORDER BY minutes_elapsed ASC;
02 / AUTOMATION WORKFLOW

From signal to mitigation, in under 90 seconds

No human in the loop on the cooldown. Analysts review after the fact, on a queue Sentinel populates with full context.

STEP 01 LIVE
◉
DB Monitor
Airflow polls the ledger every 5 minutes. Streams new deposit and withdrawal rows into the detection job.
postgres · ledger.*
STEP 02 TRIGGERED
⚡
SQL Tripwire
CTE query returns matching rows. Each match emits a structured event with user_id, ratio, and time delta.
risk_flag = PASS_THROUGH_LAYERING
STEP 03 ENFORCED
🛡
API: 72H Cooldown
POST to internal Risk API places a fiat-withdrawal hold on the account. Reversible by L2 compliance only.
POST /risk/v2/cooldown
